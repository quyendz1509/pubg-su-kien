(function() {
    console.log("%c[GODMODE] Đang thiết lập bộ lọc đánh chặn hệ thống...", "color: #3b82f6; font-weight: bold;");

    // Lưu trữ các hàm đã bind để truy ngược context
    const boundTargets = new WeakMap();

    // 1. Đánh chặn Function.prototype.bind để bắt Phaser.Game instance khi khởi tạo
    const originalBind = Function.prototype.bind;
    Function.prototype.bind = function(context, ...args) {
        const boundFn = originalBind.apply(this, [context, ...args]);
        if (context) {
            boundTargets.set(boundFn, context);
            try {
                // Kiểm tra cấu trúc đặc trưng của Phaser.Game instance
                if (context.config && context.renderer && context.loop && context.scene && !window.hackedGame) {
                    window.hackedGame = context;
                    console.log("%c[GODMODE] Đã bắt được Game Instance qua bind!", "color: #10b981; font-weight: bold;");
                }
            } catch(e) {}
        }
        return boundFn;
    };

    // 2. Đánh chặn requestAnimationFrame để bắt callback từ vòng lặp game
    const originalRAF = window.requestAnimationFrame;
    window.requestAnimationFrame = function(callback) {
        if (!window.hackedGame) {
            const target = boundTargets.get(callback);
            if (target) {
                try {
                    let gameCandidate = target.game || (target.manager && target.manager.game);
                    if (gameCandidate && gameCandidate.config && gameCandidate.renderer) {
                        window.hackedGame = gameCandidate;
                        console.log("%c[GODMODE] Đã bắt được Game Instance qua game loop!", "color: #10b981; font-weight: bold;");
                    }
                } catch(e) {}
            }
        }
        return originalRAF.apply(this, arguments);
    };

    // 3. Hàm áp dụng bất tử lên nhân vật
    const applyHack = () => {
        if (!window.hackedGame) return false;
        const game = window.hackedGame;
        if (!game.scene || !game.scene.scenes) return false;

        const activeScene = game.scene.scenes.find(s => s.player);
        if (!activeScene || !activeScene.player) return false;

        const player = activeScene.player;
        const playerProto = Object.getPrototypeOf(player);

        if (playerProto && !playerProto._isGodModeInjected) {
            playerProto.isInvincibleAt = function() { return true; };
            playerProto.takeNpcHit = function() { return false; };
            playerProto._isGodModeInjected = true;
            console.log("%c[GODMODE] KÍCH HOẠT THÀNH CÔNG! Nhân vật của bạn đã bất tử vĩnh viễn 🚀", "color: #10b981; font-weight: bold; font-size: 14px;");
            return true;
        }
        return playerProto && playerProto._isGodModeInjected;
    };

    // Kiểm tra liên tục để kích hoạt ngay khi player xuất hiện
    setInterval(applyHack, 500);

    console.log("%c[GODMODE] BỘ LỌC ĐÃ SẴN SÀNG! Vui lòng làm theo hướng dẫn bên dưới 👇", "color: #f59e0b; font-weight: bold;");
})();
